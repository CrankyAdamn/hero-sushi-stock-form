<script>
    function calculateOrder() {
        const rows = document.querySelectorAll('tbody tr:not(.section-header)');
        rows.forEach(row => {
            const inputs = row.querySelectorAll('.calc');
            const expected = parseInt(row.querySelector('td:nth-child(5)').innerText) || 0;
            const stock = Array.from(inputs).reduce((sum, input) => sum + (parseInt(input.value) || 0), 0);
            const order = Math.max(0, expected - stock);
            row.querySelector('.result').innerText = order;
        });
    }
</script>
